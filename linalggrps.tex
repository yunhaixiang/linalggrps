\section{Categories and Functors}
We start with a crash course in category theory. 

\begin{definition}A \textit{category} $\mathscr{C}$ consists of the data of
\begin{enumerate}
	\item A class\footnote{A collection of sets, which is not necessarily a set itself, such as the class of all sets.} of \textit{objects} $\mathscr{C}$, abusing of notation;
	\item For each pair of objects $X,Y\in \mathscr{C}$, a class $\mathrm{Hom}_{\mathscr{C}}(X,Y)$ of \textit{morphisms} from \textit{domain} $X$ to \textit{codomain} $Y$, denoted $f:X\rightarrow Y$, by which we mean triples\footnote{Morally the domain and codomain should be part of the data of a morphism, so that two morphisms are not equal if they do not share their domains and codomains.} $(X,Y,g)$ where $g$ is any set, called the \textit{graph} of $f$;
	\item For each triple of object $X,Y,Z\in\mathscr{C}$, a \textit{composition} function 
	\[\begin{aligned}
        \mathrm{Hom}_{\mathscr{C}}(Y,Z)\times\mathrm{Hom}_{\mathscr{C}}(X,Y)&\rightarrow \mathrm{Hom}_{\mathscr{C}}(X,Z)\\
        (f,g)&\mapsto f\circ g
    \end{aligned}\]
\end{enumerate}
such that the following axioms are satisfied
\begin{axioms}
	\item\label{cat:identity} \textit{Identity}: for each $X\in\mathscr{C}$, there exists an \textit{identity morphism} $$\mathrm{id}_X:X\rightarrow X$$ such that for any morphisms  $f:X\rightarrow Y$ and $g:Y\rightarrow X$, \[f\circ \mathrm{id}_X=f\quad\mathrm{and}\quad\mathrm{id}_X\circ g=g\]
	\item\label{cat:associativity} \textit{Associativity}: for each triple of morphisms $f:X\rightarrow Y$, $g:Y\rightarrow Z$, and $h:Z\rightarrow W$, we have \[(h\circ g)\circ f=h\circ (g\circ f)\]
\end{axioms}
\end{definition}
\begin{convention}
    Write $\mathrm{Hom}(X,Y)$ for $\mathrm{Hom}_{\mathscr{C}}(X,Y)$ when clear. 
\end{convention}
\begin{exercise}Show that the identity morphism for an object is unique.
\end{exercise}

\begin{definition}Let $f:X\rightarrow Y$ be a morphism in a category, then $f$ is a 
    \begin{enumerate}
	\item \textit{monomorphism} if $f\circ g=f\circ h$ implies $g=h$ for any $g,h:Z\rightarrow X$,
	\item \textit{epimorphism} if $g\circ f=h\circ f$ implies $g=h$ for any $g,h:Y\rightarrow Z$ 
	\item \textit{split monomorphism} if there exists $g:Y\rightarrow X$ such that $g\circ f=\mathrm{id}_X$,
	\item \textit{split epimorphism} if there exists $g:Y\rightarrow X$ such that $f\circ g=\mathrm{id}_Y$,
	\item \textit{isomorphism} if it is a split monomorphism and a split epimorphism,
	\item \textit{endomorphism} if $X=Y$,
    \item \textit{automorphism} if it is an isomorphism and an endomorphism. 
\end{enumerate}
\end{definition}

\begin{convention}For an object $X$ in a category $\mathscr{C}$, we will denote by $\mathrm{End}(X)=\mathrm{Hom}(X,X)$ the endomorphisms of $X$, and we will denote by $\mathrm{Aut}(X)$ the group of automorphisms of $X$. If there is an isomorphism $f:X\rightarrow Y$ in $\mathscr{C}$, we say $X$ and $Y$ are isomorphic and write $X\cong Y$. 
\end{convention}
\begin{remark}It's an easy exercise to show split monos (resp. split epis) are monos (resp. epis).
    Moreover, suppose $\mathscr{C}$ is a category where one can talk about injective and surjective morphisms, in general, split mono (resp. split epi) is a strictly stronger condition than injective (resp. surjective), and injective (resp. surjective) is a strictly stronger condition than mono (resp. epi). 
\end{remark}


\begin{definition}
    Let $\mathscr{C}, \mathscr{D}$ be categories, a \textit{covariant functor} (or just simply a \textit{functor}) from $\mathscr{C}$ to $\mathscr{D}$, denoted $F:\mathscr{C}\rightarrow \mathscr{D}$, is the collection of the following data
    \begin{enumerate}
        \item for each object $X\in\mathscr{C}$, an object $F(X)\in \mathscr{D}$
        \item for each morphism $f:X\rightarrow Y$ in $\mathscr{C}$, a morphism $F[f]:F(X)\rightarrow F(Y)$, such that 
        \begin{itemize}
            \item $F[\mathrm{id}_X]=\mathrm{id}_{F(X)}$ for each $X\in\mathscr{C}$,
            \item $F[f\circ g]=F[f]\circ F[g]$ for each $f:X\rightarrow Y$ and $g:Y\rightarrow Z$ in $\mathscr{C}$.
        \end{itemize}
    \end{enumerate}
     A \textit{contravariant functor} from $\mathscr{C}$ to $\mathscr{D}$ is a covariant functor $F:\mathscr{C}^{\mathrm{op}}\rightarrow\mathscr{D}$. 
\end{definition}
\begin{definition}Suppose $\mathscr{C}, \mathscr{D}, \mathscr{E}$ are categories and $F:\mathscr{C}\rightarrow \mathscr{D}$ and $G:\mathscr{D}\rightarrow \mathscr{E}$ are functors, we define their \textit{composite functor} $G\circ F:\mathscr{C}\rightarrow\mathscr{E}$ as the functor that maps $X\mapsto G(F(X))$ for objects and maps the morphisms by $(G\circ F)[f]=G[F[f]]$ for each $f:X\rightarrow Y$. 
\end{definition}
\begin{definition}Let $\mathscr{C}$ be a category, then the \textit{identity functor} $\mathrm{id}_{\mathscr{C}}:\mathscr{C}\rightarrow \mathscr{C}$ for $\mathscr{C}$ is the functor that maps each object $X\in\mathscr{C}$ by $\mathrm{id}_{\mathscr{C}}(X)=X$ and each morphism $f:X\rightarrow Y$ by $\mathrm{id}_{\mathscr{C}}[f]=f$. 
\end{definition}

\begin{example}Here are some examples of functors in nature \label{eg:func}
    \begin{enumerate}
        \item the functor $\mathsf{Grp}\rightarrow \mathsf{Set}$ that maps a group to its underlying set and sends morphisms to themselves (functor that ``forget'' data such as this are called \textit{forgetful functors})
        \item the functor $(-)^\times:\mathsf{Ring}\rightarrow\mathsf{Grp}$ which sends a ring $R$ to its multiplicative group of units, and sends a morphism of rings to its restriction on the groups of units,
        \item the functor $\mathrm{GL}_n(-):\mathsf{Ring}\rightarrow\mathsf{Grp}$ which sends a ring $R$ to $\mathrm{GL}_n(R)$ the group of invertible matrices in $R$, and send a morphism to the obvious entry-wise group homomorphism. 
        \item the functor $(-\otimes_R M):\mathsf{Mod}_R\rightarrow \mathsf{Mod}_R$ for an $R$-module $M$, which sends a $R$-module $N$ to $N\otimes_R M$ and a morphism $f:N\rightarrow P$ to $f\otimes \mathrm{id}_M$,
        \item the contravariant functor $(-)^*:\mathsf{Mod}_R^{\mathrm{op}}\rightarrow \mathsf{Mod}_R$ for a ring $R$ which sends an $R$-module $M$ to its dual module $M^*$, and a morphism $f^{\mathrm{op}}:M\rightarrow N$ to $f^*:N^*\rightarrow M^*$ by $g\mapsto g\circ f$, 
        \item the functor $\pi_1(-):\mathsf{PCTop}\rightarrow\mathsf{Grp}$ where $\mathsf{PCTop}$ is the full subcategory of $\mathsf{Top}$ of path connected spaces, which sends a space $X$ to its fundamental group $\pi_1(X)$ and a continuous map $f:X\rightarrow Y$ to its induced map on fundamental groups $f_*:\pi_1(X)\rightarrow \pi_1(Y)$. 
    \end{enumerate}
\end{example}

\begin{definition}
    Define the \textit{opposite category} $\mathscr{C}^{\mathrm{op}}$ of a category $\mathscr{C}$ to be the category with the same objects as $\mathscr{C}$ but with its morphisms $\mathrm{Hom}_{\mathscr{C}^{\mathrm{op}}}(X,Y)=\mathrm{Hom}_{\mathscr{C}}(Y,X)$ for $X,Y\in \mathscr{C}^{\mathrm{op}}$.
    For each $f:X\rightarrow Y$ in $\mathscr{C}$, we denote by $f^{\mathrm{op}}:Y\rightarrow X$ the corresponding \textit{opposite morphism} in $\mathscr{C}^{\mathrm op}$, that is, it has the same mapping as $f$, but has its domain and codomain swapped. 
\end{definition}

\begin{definition}A \textit{subcategory} of a category $\mathscr{C}$ is a category $\mathscr{D}$ such that its objects $\mathscr{D}\subseteq \mathscr{C}$ is a subcollection, its morphisms $\mathrm{Hom}_{\mathscr{D}}(X,Y)\subseteq \mathrm{Hom}_{\mathscr{C}}(X,Y)$ for any $X,Y\in\mathscr{D}$, and the composition function of $\mathscr{D}$ is the restriction of that of $\mathscr{C}$. We say $\mathscr{D}$ is a \textit{full subcategory} if further that  $\mathrm{Hom}_{\mathscr{D}}(X,Y)= \mathrm{Hom}_{\mathscr{C}}(X,Y)$ for any $X,Y\in\mathscr{D}$. 
\end{definition}
\begin{definition}Suppose $\mathscr{C},\mathscr{D}$ are categories, their product category $\mathscr{C}\times \mathscr{D}$ is the category where objects are pairs $(X,Y)$ where $X\in \mathscr{C}$ and $Y\in \mathscr{D}$, and morphisms are 
    \[\mathrm{Hom}((X_1,Y_1),(X_2,Y_2))=\mathrm{Hom}(X_1,X_2)\times \mathrm{Hom}(Y_1,Y_2)\]
    with element-wise composition. 
\end{definition}



\begin{definition}
    We say that the category $\mathscr{C}$ is \textit{locally small} if $\mathrm{Hom}(X,Y)$ is a set for all $X$ and $Y$, and \textit{small} if it is locally small and the class of objects of $\mathscr{C}$ is also a set. 
\end{definition}

\begin{definition}
    \label{def:hom-functors}
    Suppose $\mathscr{C}$ is locally small and $X\in\mathscr{C}$. Define the \textit{(covariant) hom-functor} of $X$
    \[\begin{aligned}
    \mathrm{Hom}(X,-):\mathscr{C}&\rightarrow \mathsf{Set}\\
        Y&\mapsto \mathrm{Hom}(X,Y)\\
        (f:Y\rightarrow Z)&\mapsto (f\circ -)
    \end{aligned}
    \] 
    where $(f\circ -):\mathrm{Hom}(X,Y)\rightarrow \mathrm{Hom}(X,Z)$ maps $g\mapsto f\circ g$. The \textit{contravariant hom-functor} of $X$ is 
    \[
    \begin{aligned}
    \mathrm{Hom}(-,X):\mathscr{C}^{\mathrm{op}}&\rightarrow \mathsf{Set}\\
        Y&\mapsto \mathrm{Hom}(Y,X)\\
        (f^{\mathrm{op}}:Z\rightarrow Y)&\mapsto (-\circ f)
    \end{aligned}
    \] 
    where $(-\circ f):\mathrm{Hom}(Z,X)\rightarrow \mathrm{Hom}(Y,X)$ maps $g\mapsto g\circ f$. The \textit{bivariate hom-functor} of $X$ is 
    \[\begin{aligned}
        \mathrm{Hom}(-,-):\mathscr{C}^{\mathrm{op}}\times \mathscr{C}&\rightarrow \mathscr{D}\\
         (X,Y)&\mapsto \mathrm{Hom}(X,Y)\\
         (f^{\mathrm{op}},g):(X_1,Y_1)\rightarrow (X_2,Y_2)&\mapsto (g\circ -\circ f)
    \end{aligned}\]
    where 
    $(g\circ -\circ f):\mathrm{Hom}(X_1,Y_1)\rightarrow \mathrm{Hom}(X_2,Y_2)$ maps $h\mapsto g\circ h\circ f$
\end{definition}

\begin{remark}Suppose $\mathscr{C}$ is locally small and $X\in\mathscr{C}$. We also use the notations
    \[h^X:=\mathrm{Hom}(X,-)\qquad h_X:=\mathrm{Hom}(-,X)\qquad h:=\mathrm{Hom}(-,-)\]
    for the covariant, contravariant, and bivariate hom-functors. 
\end{remark}

\begin{definition}Let $\mathscr{I},\mathscr{C}$ be categories, a \textit{diagram} indexed by $\mathscr{I}$ in $\mathscr{C}$ is simply a functor $F:\mathscr{I}\rightarrow \mathscr{C}$. We say that the diagram $F$ \textit{commutes} if for each $X,Y\in\mathscr{I}$, if $f,g\in\mathrm{Hom}(X,Y)$ then $F[f]=F[g]$. 
\end{definition}


\begin{example}We represent a diagram $F:\mathscr{I}\rightarrow \mathscr{C}$ as a directed multigraph in the same shape as the index category $\mathscr{I}$, and label the vertices and edges with their images of $F$ in $\mathscr{C}$. For example,
    \[\begin{tikzcd}[sep=large]
	A & B \\
	C & D
	\arrow["f", from=1-1, to=1-2]
	\arrow["h"', from=1-1, to=2-1]
	\arrow["g", from=1-2, to=2-2]
	\arrow["e"', from=2-1, to=2-2]
\end{tikzcd}\]
and this square commutes if and only if $g\circ f=e\circ h$. 
\end{example}

    This is straightforward from the definition of commutative diagrams.
\end{remark}

\begin{definition}Let $\mathscr{C},\mathscr{D}$ be categories, and $F,G:\mathscr{C}\rightarrow \mathscr{D}$ be functors, a \textit{natural transformation} from $F$ to $G$, denoted $\varphi:F\Rightarrow G$, is the data of a morphism $\varphi_X:F(X)\rightarrow G(X)$ for each $X\in \mathscr{C}$ such that the following diagram commutes
\[\begin{tikzcd}[sep=large]
	F(X) & G(X) \\
	F(Y) & G(Y)
	\arrow["\varphi_X", from=1-1, to=1-2]
	\arrow["F {[ f ]} "', from=1-1, to=2-1]
	\arrow["G {[ f ]}", from=1-2, to=2-2]
	\arrow["\varphi_Y"', from=2-1, to=2-2]
\end{tikzcd}\]
for all morphisms $f:X\rightarrow Y$ in $\mathscr{C}$. In other words, $G[f]\circ \varphi_X=\varphi_Y\circ F[f]$ for all $f:X\rightarrow Y$ in $\mathscr{C}$. 
Moreover, if $\varphi_X$ is an isomorphism for each $X\in\mathscr{C}$, we say $\varphi$ is a \textit{natural isomorphism}. 
\end{definition}

\begin{definition}Let $\mathscr{C},\mathscr{D}$ be categories, and let $E,F,G:\mathscr{C}\rightarrow\mathscr{D}$ be functors. Suppose $\varphi:E\Rightarrow F$ and $\psi:F\Rightarrow G$  are natural transformations. We define their \textit{(vertical) composition} $\psi\circ \varphi:E\Rightarrow G$ as the natural transformation $(\psi\circ \varphi)_X=\psi_X\circ \varphi_X$ for every $X\in\mathscr{C}$. 
\end{definition}

\begin{definition}Let $\mathscr{C},\mathscr{D}$ be categories, the \textit{functor category} from $\mathscr{C}$ to $\mathscr{D}$, denoted $\mathrm{Fun}(\mathscr{C},\mathscr{D})$, is the category where objects are functors $F:\mathscr{C}\rightarrow \mathscr{D}$, morphisms are natural transformations, and composition is given by (vertical) composition of natural transformations. 
\end{definition}

\begin{remark}
    \label{rmk:isom}
    Natural isomorphisms are precisely the isomorphisms in the functor category. 
\end{remark}

\begin{definition}Let $\mathscr{C},\mathscr{D}$ be categories, a functor $F:\mathscr{C}\rightarrow \mathscr{D}$ is called an \textit{equivalence} if there is a functor $G:\mathscr{D}\rightarrow\mathscr{C}$ such that there are natural isomorphisms $\eta:\mathrm{id}_{\mathscr{C}}\Rightarrow G\circ F$ and $\varepsilon:F\circ G\Rightarrow\mathrm{id}_{\mathscr{D}}$. 
    If there is an equivalence between $\mathscr{C}$ and $\mathscr{D}$, we say they are \textit{equivalent} and write $\mathscr{C} \simeq \mathscr{D}$. Moreover, we call $G$ the \textit{quasi-inverses} of $F$ and call the pair of functors an \textit{equivalence of categories}. 
\end{definition}

\begin{example}Here are some examples of equivalent categories
    \begin{enumerate}
        \item The category $\mathsf{Vect}_K$ of finite dimensional $K$-vector spaces is equivalent to the category where objects are $K^n$ for $n\ge 0$ and morphisms $\mathrm{Hom}(K^n,K^m)$ are the matrices $\mathrm M_{n,m}(K)$. 
        \item The category of intemediate extensions of a finite Galois extension $L/K$ with $L$-embeddings as morphisms, is equivalent to the category of finite transitive $G$-sets where $G=\mathrm{Gal}(L/K)$. 
        \item The category of complex representations of a finite group $G$ (whose morphisms are equivariant maps) is equivalent to the category of $\mathbb C[G]$-modules, and they are also equivalent to $\mathrm{Fun}(\mathrm BG,\mathsf{Vect}_{\mathbb C})$, where $\mathrm B G$ is the category with a single object $\bullet$, with the endomorphisms of $\bullet$ being group elements, and with the group operation as its composition. 
        \item R-algebras and ring maps
    \end{enumerate}
    It is an easy exercise for the reader to spot what the unstated equivalences in the cases above are.
\end{example}

\begin{definition}Suppose $\mathscr{C}, \mathscr{D}$ are categories and $F:\mathscr{C}\rightarrow \mathscr{D}$ is a functor. 
    For each $X,Y\in\mathscr{C}$, then
    \[\mathrm{Hom}(X,Y)\xrightarrow{f\mapsto F[f]} \mathrm{Hom}(F(X),F(Y))\]
    is a map of sets. We say that $F$ is 
    \begin{enumerate}
        \item \textit{faithful} if the above map is injective for all $X,Y\in\mathscr{C}$,
        \item \textit{full} if the above map is surjective for all $X,Y\in\mathscr{C}$,
        \item \textit{fully faithful} if the above map is bijective for all $X,Y\in\mathscr{C}$,
    \end{enumerate}
    Moreover, call $F$ \textit{essentially surjective} if for all $Y\in \mathscr{D}$ exists $X\in\mathscr{C}$ such that $F(X)\cong Y$ in $\mathscr{D}$. 
\end{definition}
\begin{theorem}A functor is an equivalence iff it is fully faithful and essentially surjective. 
    \begin{proof}
        Technical. See \cite[Thm.~1.3.13, p.~22]{KashiwaraSchapira2006}.
    \end{proof}
\end{theorem}

\begin{definition}Let $\mathscr{C}$ be a locally small category. Define 
    \[\widehat{\mathscr{C}}:=\mathrm{Fun}(\mathscr{C}^{\mathrm{op}},\mathsf{Set})\qquad\widecheck{\mathscr{C}}:=\mathrm{Fun}(\mathscr{C},\mathsf{Set})\] 
    as the \textit{category of presheaves} of $\mathscr{C}$ and \textit{category of copresheaves} of $\mathscr{C}$ respectively.
    The \textit{Yoneda embedding} is the contravariant functor 
    \[\begin{aligned}
        h^{\bullet}:\mathscr{C}^{\mathrm{op}}&\rightarrow \widecheck{\mathscr{C}}\\
        X&\mapsto h^X=\mathrm{Hom}(X,-)\\
        (f^{\mathrm{op}}:Z\rightarrow Y)&\mapsto \left[X\mapsto (-\circ f)\right]
    \end{aligned}\]
    where $(-\circ f):\mathrm{Hom}(Z,X)\rightarrow \mathrm{Hom}(Y,X)$ is as in \Cref{def:hom-functors}. 
    Dually, the \textit{co-Yoneda embedding} is
    \[\begin{aligned}
        h_{\bullet}:\mathscr{C}&\rightarrow \widehat{\mathscr{C}}\\
        X&\mapsto h_X=\mathrm{Hom}(-,X)\\
        (f:Y\rightarrow Z)&\mapsto \left[X\mapsto (f\circ -)\right]
    \end{aligned}\]
    where $(f\circ -):\mathrm{Hom}(X,Y)\rightarrow \mathrm{Hom}(X,Z)$ is as in \Cref{def:hom-functors}.
    
\end{definition}


\begin{lemma}[Yoneda]
    \label{thm:yoneda}
    Let $\mathscr{C}$ be a locally small category. Suppose $X\in\mathscr{C}$ and $F:\mathscr{C}\rightarrow\mathsf{Set}$ is a functor, then there are bijections $\Phi$ and $\Psi$, shown below, 
    \[\begin{tikzcd}[column sep=3em] % try 6em, 8em, 12em, or even 3cm
        {\mathrm{Hom}_{\widecheck{\mathscr{C}}}(h^{X},F)}
        \arrow[rightarrow, r, bend left=30, "\Phi"{yshift=2pt}]
        & {F(X)}
        \arrow[rightarrow, l, bend left=30, "\Psi"{yshift=-2pt}]
    \end{tikzcd}
    \]
    which are inverses of each other.

    \begin{proof}
        For a natural transformation $\varphi:h^X\Rightarrow F$ and a morphism $f:X\rightarrow Y$ in $\mathscr{C}$, the diagram 
        \[
        \begin{tikzcd}[sep = huge]
        \mathrm{Hom}_{\mathscr{C}}(X,X) \arrow[r, "\varphi_X"] \arrow[d, "(f\circ - )"'] & F(X) \arrow[d, "F{[f]}"] \\
        \mathrm{Hom}_{\mathscr{C}}(X,Y) \arrow[r, "\varphi_Y"]                 & F(Y)               
        \end{tikzcd}\]
        commutes. Thus $\varphi_Y\circ (f\circ -)=F[f]\circ \varphi_X$. Evaluating at $\mathrm{id}_X$ on both sides yields the identity
        \[\varphi_Y(f)=F[f](\varphi_X(\mathrm{id}_X))\]
        therefore the natural transformation $\varphi$ is completely determined by $u=\varphi_X(\mathrm{id}_X)$. Therefore, let
        \[\Phi:\mathrm{Hom}_{\widecheck{\mathscr{C}}}(h^{X},F)\rightarrow F(X)\qquad \Phi(\varphi)=\varphi_X(\mathrm{id}_X)\]
        and, by the same identity, the inverse of $\Phi$ is naturally
        \[\Psi:F(X)\rightarrow \mathrm{Hom}_{\widecheck{\mathscr{C}}}(h^{X},F)\qquad \Psi(u)=\varphi\]
        where $\varphi_Y:\mathrm{Hom}(X,Y)\rightarrow F(Y)$ is given by $\varphi_Y(f)=(F[f])(u)$, for each $Y\in \mathscr{C}$. It is not hard for the reader to verify that indeed $\Psi$ and $\Phi$ are inverses of each other.
    \end{proof}
\end{lemma}
\begin{remark}
    \label{rmk:yoneda}
    The preceding \Cref{thm:yoneda} is known as \textit{Yoneda lemma} or the \textit{fundamental theorem of category theory}. 
    Notice that the definition of $\Phi$ and $\Psi$ does not involve in $X$ and $F$, therefore one can upgrade the lemma to say that there is a mutually inverse pair of natural isomorphism 
    \[\begin{tikzcd}[column sep=3em] % try 6em, 8em, 12em, or even 3cm
        {\mathrm{Hom}_{\widecheck{\mathscr{C}}}(h^{\bullet},-)}
        \arrow[Rightarrow, r, bend left=30, "\Phi"{yshift=2pt}]
        & {(-)(\bullet)}
        \arrow[Rightarrow, l, bend left=30, "\Psi"{yshift=-2pt}]
    \end{tikzcd}
    \]
    where the left hand side functor $\mathrm{Hom}_{\widecheck{\mathscr{C}}}(h^{\bullet},-):\mathscr{C}\times\widecheck{\mathscr{C}}\rightarrow\mathsf{Set}$ is given by the composition 
    \[\mathscr{C}\times \widecheck{\mathscr{C}}\xrightarrow{h^\bullet\times \mathrm{id}_{\widecheck{\mathscr{C}}}} \widecheck{\mathscr{C}}\times \widecheck{\mathscr{C}}\xrightarrow{\mathrm{Hom}_{\widecheck{\mathscr{C}}}(-,-)}\mathsf{Set}\]
    and the right hand side functor, called the \textit{evaluation functor}, is given by 
    \[(-)(\bullet):\mathscr{C}\times \widecheck{\mathscr{C}}\rightarrow\mathsf{Set}\qquad (X,F)\mapsto F(X) \]
    and sends a morphism $(f:X\rightarrow Y,\varphi:F\Rightarrow G)$ to the morphism $\varphi_Y\circ F[f]=G[f]\circ \varphi_X$. Same constructions for $\Phi$ and $\Psi$ will work, hence we have shown that the Yoneda lemma is \textit{functorial} or \textit{natural} in the variables $X\in\mathscr{C}$ and $F\in \widecheck{\mathscr{C}}$. 
\end{remark}

\begin{exercise}Prove a dual and functorial version of the Yoneda lemma: let $\mathscr{C}$ be a locally small category. Suppose $X\in\mathscr{C}$ and $F:\mathscr{C}^{\mathrm{op}}\rightarrow\mathsf{Set}$, then construct natural isomorphisms
    \[\begin{tikzcd}[column sep=3em] % try 6em, 8em, 12em, or even 3cm
        {\mathrm{Hom}_{\widehat{\mathscr{C}}}(h_{\bullet},-)}
        \arrow[Rightarrow, r, bend left=30, "\Pi"{yshift=2pt}]
        & {(-)(\bullet)}
        \arrow[Rightarrow, l, bend left=30, "\Theta"{yshift=-2pt}]
    \end{tikzcd}
    \]
    which are inverses of each other, where you should define the functor $\mathrm{Hom}_{\widehat{\mathscr{C}}}(h_{\bullet},-):\mathscr{C}\times\widehat{\mathscr{C}}\rightarrow\mathsf{Set}$ and the \textit{coevaluation functor} $(-)(\bullet):\mathscr{C}\times \widehat{\mathscr{C}}\rightarrow\mathsf{Set}$ dually to those in \Cref{rmk:yoneda}. 
\end{exercise}
\begin{theorem}\label{thm:yff}Let $\mathscr{C}$ be a locally small category, then the Yoneda and co-Yoneda embeddings 
    \[h^\bullet: \mathscr{C}^{\mathrm{op}}\rightarrow \widecheck{\mathscr{C}}\quad\mathrm{and}\quad h_\bullet: \mathscr{C}\rightarrow \widehat{\mathscr{C}}\]
    are fully faithful. 
    \begin{proof}
        We will only show $h^\bullet$ is fully faithful: for the dual $h_\bullet$, the proof is completely symmetric. 
        Let $X,Y\in\mathscr{C}$. By the Yoneda lemma \Cref{thm:yoneda}, there is a bijection 
        \[\Phi:\mathrm{Hom}_{\widecheck{\mathscr{C}}}(h^X, h^Y)\rightarrow h^Y(X)=\mathrm{Hom}_{\mathscr{C}}(Y,X)\]
        by specifying $F=h^Y$. One can verify that this gives the inverse for the map
        \[\mathrm{Hom}_{\mathscr{C}}(Y,X)=\mathrm{Hom}_{\mathscr{C}^{\mathrm{op}}}(X,Y)\xrightarrow{f\mapsto h^\bullet\left[f\right]} \mathrm{Hom}_{\widecheck{\mathscr{C}}}(h^X, h^Y)\]
        Thus $h^\bullet$ is fully faithful.\end{proof}
\end{theorem}


\begin{lemma}\label{lem:isom}Let $\mathscr{C},\mathscr{D}$ be categories and $F:\mathscr{C}\rightarrow\mathscr{D}$ a fully faithful functor, then the morphism  $F[f]:F(X)\rightarrow F(Y)$ is an isomorphism in $\mathscr{D}$ if and only if $f:X\rightarrow Y$ is an isomorphism in $\mathscr{C}$. 
    \begin{proof}
    Assume $f:X\rightarrow Y$ is an isomorphism in $\mathscr{C}$, then there exists $g:Y\rightarrow X$ with $g\circ f=\mathrm{id}_X$ and $f\circ g=\mathrm{id}_Y$.
    Applying functor $F$, we get $F[g]\circ F[f]=\mathrm{id}_{F(X)}$ and $F[f]\circ F[g]=\mathrm{id}_{F(Y)}$, hence $F[f]$ has inverse $F[g]$, which makes it an isomorphism in $\mathscr{D}$.
    Conversely, by fullness of $F$, choose $g:Y\rightarrow X$ with $F[g]=F[f]^{-1}$. Then we have $F[f\circ g]=F[f]\circ F[g]=\mathrm{id}_{F(X)}$ so by faithfulness of $F$, we have $f\circ g=\mathrm{id}_X$. By symmetric arguments, we have $g\circ f=\mathrm{id}_Y$, so $g$ is the inverse of $f$.  
    Therefore $f$ is an isomorphism in $\mathscr{C}$. 
    \end{proof}
\end{lemma}
\begin{theorem}
    \label{thm:yon}
    Let $\mathscr{C}$ be a locally small category, then for a morphism $f:X\rightarrow Y$ in $\mathscr{C}$, TFAE
    \begin{enumerate}
        \item $f:X\rightarrow Y$ is an isomorphism in $\mathscr{C}$,
        \item $h_\bullet[f]:h_X\rightarrow h_Y$ is an isomorphism in $\widehat{\mathscr{C}}$,
        \item $h^\bullet[f^{\mathrm{op}}]:h^Y\rightarrow h^X$ is an isomorphism in $\widecheck{\mathscr{C}}$,
    \end{enumerate}
    \begin{proof}
        Straightforward by \Cref{thm:yff} and \Cref{lem:isom}. 
    \end{proof}
\end{theorem}

\begin{remark}What \Cref{thm:yff} and \Cref{thm:yon} tells us is that some presheaves (resp. copresheaves) on a category can be represented as objects in the category, namely these are the ones in the image of the Yoneda (resp. co-Yoneda) embedding. 
\end{remark}
\begin{definition}Let $\mathscr{C}$ be a locally small category, we call a presheaf $F\in \widecheck{\mathscr{C}}$ (resp. copresheaf $F\in \widehat{\mathscr{C}}$) \textit{representable} by an object $X\in\mathscr{C}$ if $h^X\cong F$ (resp. $h_X\cong F$). 
\end{definition}


\section{Schemes and Group Schemes}

Unless specified otherwise, fix a ring $R$, which we shall call our \textit{ground ring}. 
\begin{convention}
    We assume rings are commutative and unital, and algebras are associative, commutative, and unital. 
    In particular, in our convention an algebra $A$ over a ring $R$ is equivalent to the data of a ring $A$ with a ring map $\varphi:R\rightarrow A$, viewing the scalar multiplication as $r\cdot a=\varphi(r)a$. 
    We will use this equivalence frequently and often implicitly. 
\end{convention}

\begin{definition}A \textit{$R$-functor} is a functor \[X:\mathsf{Alg}_R\rightarrow \mathsf{Set}\]
    and the \textit{category of $R$-functors} is the functor category $\mathsf{Fun}_R:=\mathrm{Fun}(\mathsf{Alg}_R,\mathsf{Set})$.
\end{definition}
\begin{definition}
Suppose $A$ is a $R$-algebra, its \textit{spectrum} is the $R$-functor 
    \[\mathrm{Spec}(A):=\mathrm{Hom}_{\mathsf{Alg}_R}(A,-):\mathsf{Alg}_R\rightarrow \mathsf{Set}\]
    An \textit{affine $R$-scheme} is the spectrum of a $R$-algebra. The \textit{affine $n$-space} is the affine $R$-scheme \[\mathbb{A}^n_R:=\mathrm{Spec}(R[x_1,\dots,x_n])\]
    The category $\mathsf{Aff}_R$ of affine $R$-schemes is the full subcategory of all affine $R$-schemes of $\mathsf{Fun}_R$. 
\end{definition}
\begin{remark}
The notion of affine $R$-schemes generalizes the notion of affine varieties. 
Readers who have experience in algebraic geometry might have been taught that an affine $K$-variety for some algebraically closed field $K$ is a subset of $K^n$ for some $n$ given by the vanishing set of some polynomials $f_1,\dots,f_m\in K[x_1,\dots,x_n]$, that is, the set
\[\{(x_1,\dots,x_n)\in K^n:f_1(x_1,\dots,x_n)=\cdots=f_m(x_1,\dots,x_n)=0\}\]
For our purposes, however, we take a \textit{functorial} perspective to varieties. To see what this is about, we first make the important observation that the above set can be naturally identified as the set
\[X(K)=\mathrm{Hom}_{\mathsf{Alg}_K}\left(\frac{K[x_1,\dots,x_n]}{(f_1,\dots,f_m)},K\right)\quad\mathrm{where}\quad X=\mathrm{Spec}\left(\frac{K[x_1,\dots,x_n]}{(f_1,\dots,f_m)}\right)\]
Namely, each $\phi\in X(K)$ can be identified with $(\phi(x_1),\dots,\phi(x_n))\in K^n$. 
Similarly, for polynomials $f_1,\dots,f_m\in R[x_1,\dots,x_n]$, the set 
\[X(S)=\mathrm{Hom}_{\mathsf{Alg}_R}\left(\frac{R[x_1,\dots,x_n]}{(f_1,\dots,f_m)},S\right)\quad\mathrm{where}\quad X=\mathrm{Spec}\left(\frac{R[x_1,\dots,x_n]}{(f_1,\dots,f_m)}\right)\]
for an $R$-algebra $S$ can be identified as the set 
\[\{(x_1,\dots,x_n)\in S^n:f_1(x_1,\dots,x_n)=\cdots=f_m(x_1,\dots,x_n)=0\}\]
where $f_1,\dots,f_m$ are identified as their images along the natural map $R[x_1,\dots,x_n]\rightarrow S[x_1,\dots,x_n]$ induced by the structure map $R\rightarrow S$. 
Thus, what the spectrum of a $R$-algebra encodes is the vanishing sets of a set of polynomials over each $R$-algebra. 

\end{remark}
\begin{definition}Suppose that $X=\mathrm{Spec}(A)$ is an affine $R$-scheme where $A$ is a $R$-algebra. Let $S$ be another $R$-algebra then we call the set 
    \[X(S)=\mathrm{Hom}_{\mathsf{Alg}_R}(A,S)\]
    the \textit{$S$-points} of $X$. When $S$ is a field, it is also called the \textit{$S$-rational points} of $X$. 
\end{definition}

\begin{remark}
Recall that a $R$-algebra $A$ is said to be finitely generated if there is a surjective map $R[x_1,\dots,x_n]\rightarrow A$, or equivalently $A=R[x_1,\dots,x_n]/I$ for some ideal $I\subseteq K[x_1,\dots,x_n]$. 
If $R$ is noetherian, that is, each of its ideals is finitely generated, then by Hilbert's basis theorem, so is $R[x_1,\dots,x_n]$, whence the finitely generated $R$-algebras are  of the form $R[x_1,\dots,x_n]/(f_1,\dots,f_m)$. 

\end{remark}

\begin{remark}
One huge advantage that affine schemes provide is their treatment of \textit{nilpotence}. The vanishing set of a polynomial $f\in K[x_1,\dots,x_n]$ for $K$ a field is the same as that of $f^2$. However, 
\[\mathrm{Spec}(K[x_1,\dots,x_n]/(f))\ne \mathrm{Spec}(K[x_1,\dots,x_n]/(f^2))\]
with the right hand side being thought of as having an \textit{infinitesimal thickening} or \textit{nilpotent thickening}. 

\end{remark}

\begin{remark}Let $S$ be a $R$-algebra, then $\mathbb{A}^1_R(S)=\mathrm{Hom}_{\mathsf{Alg}_R}(R[x],S)$ is canonically a $R$-algebra by
    \[\begin{aligned}
        (\phi+\psi)(f)&=\phi(f)+\psi(f)\\
        (\phi \psi)(f)&=\phi(f)\psi(f)\\
        (r\phi)(f)&=r\phi(f)
    \end{aligned}
    \]
    for each $\phi,\psi\in \mathbb A_R^1(S)$ and $r\in R$, where $f\in R[x]$. Note that naturally $\mathbb A_R^1(S)\cong S$ as $R$-algebras. 
\end{remark}
\begin{definition}
    For an affine $R$-scheme $X:\mathsf{Alg}_R\rightarrow\mathsf{Set}$, its \textit{coordinate ring} (resp. \textit{coordinate algebra}) is the ring (resp. $R$-algebra)
    \[O(X):=\mathrm{Hom}_{\mathsf{Aff}_R}(X,\mathbb A_R^1)\]
    with the algebraic operations given as follows: given $f,g\in O(X)$ and $r\in R$
    \[\begin{aligned}
        (f+g)_S(\phi)&=f_S(\phi)+g_S(\phi)\\
        (fg)_S(\phi)&=f_S(\phi)f_S(\phi)\\
        (rf)_S(\phi)&=rf_S(\phi)\\
    \end{aligned}
    \]
    where $\phi\in X(S)$, for each $R$-algebra $S$. Moreover, define the contravariant functor 
    \[O:\mathsf{Aff}_R^{\mathrm{op}}\rightarrow \mathsf{Alg}_R\qquad O=\mathrm{Hom}_{\mathsf{Aff}_R}(-,\mathbb A_R^1)\]
    where for a morphism $\varphi:X\rightarrow Y$ of affine $R$-schemes, the induced map
    \[O[\varphi^{\mathrm{op}}]=\varphi^{\#}:O(Y)\rightarrow O(X)\qquad \psi\mapsto \psi\circ \varphi \]
    is called the \textit{induced regular map} of $\varphi$. 
\end{definition}

\begin{definition}
Define the contravariant functor \[\mathrm{Spec}:\mathsf{Alg}_R^{\mathrm{op}}\rightarrow \mathsf{Aff}_R\qquad A\mapsto \mathrm{Spec}(A)\]
    where for each morphism $\varphi:A\rightarrow B$ of $R$-algebras, there is the induced natural transformation \[\varphi^*:\mathrm{Spec}(B)\rightarrow\mathrm{Spec}(A)\] called the \textit{pullback} given by 
    \[\varphi^*_S:\mathrm{Spec}(B)(S)\rightarrow \mathrm{Spec}(A)(S)\qquad \psi\mapsto \psi\circ\varphi\]
    for each $R$-algebras $S$. 
\end{definition}

\begin{theorem}The pair of functors $\mathrm{Spec}$ and $O$ is an equivalence of categories
\[\begin{tikzcd}
	{\mathsf{Alg}_R^{\mathrm{op}}} && {\mathsf{Aff}_R}
	\arrow["{\mathrm{Spec}}", bend left = 25, from=1-1, to=1-3]
	\arrow["{O}", shift left=1, bend left = 25, from=1-3, to=1-1]
\end{tikzcd}\]
\begin{proof}Suppose $A$ is a $R$-algebra, then applying \Cref{thm:yoneda}, namely, the Yoneda lemma
    \[O(\mathrm{Spec}(A))=\mathrm{Hom}_{\mathsf{Aff}_R}(h^{A},\mathbb{A}^1_R)\cong \mathbb{A}^1_R(A)\cong A\]
    where we leave it to the reader to check that the relevant bijection is an isomorphism of $K$-algebras. Conversely, let $X=\mathrm{Spec}(A)$, then there is an induced natural isomorphism of hom-functors
    \[\mathrm{Spec}(O(X))=\mathrm{Hom}_{\mathsf{Alg}_R}(O(\mathrm{Spec}(A)),-)\cong \mathrm{Hom}_{\mathsf{Alg}_R}(A,-)=X\]
    by \Cref{thm:yon}. Therefore $\mathrm{Spec}$ and $O$ are quasi-inverses of each other. 
\end{proof}
\end{theorem}
\begin{definition}A \textit{$R$-group functor} is a functor \[G:\mathsf{Alg}_R\rightarrow\mathsf{Grp}\] 
    The \textit{category of $R$-group functors} is the functor category $\mathsf{GFun}_R=\mathrm{Fun}(\mathsf{Alg}_R,\mathsf{Grp})$.
\end{definition}


\begin{definition}
    An \textit{affine $R$-group scheme} is a $R$-group functor $G:\mathsf{Alg}_R\rightarrow\mathsf{Grp}$ such that 
    its composition with the forgetful functor to $\mathsf{Set}$
    \[\widetilde{G}:\mathsf{Alg}_R\xrightarrow{G} \mathsf{Grp}\rightarrow\mathsf{Set}\]
    is isomorphic to an affine $R$-scheme. Further, we say an affine $R$-group scheme $G$ is an \textit{(affine) algebraic group} over $R$ if $\widetilde{G}$ is \textit{of finite type}, i.e. it is the spectrum of a finitely generated $R$-algebra. Define $\mathsf{GAff}_R$ and $\mathsf{GAlg}_R$, the categories of affine $R$-group schemes and (affine) algebraic groups over $R$, as full subcategories of $\mathsf{GFun}_R$ respectively. 
\end{definition}
\begin{convention}
    In this text, ``algebraic group'' always means affine algebraic group. However, it is important to keep in mind that there are things like abelian varieties which are considered algebraic groups in broader context but are by no means affine.
\end{convention}

\begin{definition}Let $S$ be an $R$-algebra.

\end{definition}

\begin{convention}Let $X=(x_{i,j})_{1\le i,j\le n}$ be a $n$-by-$n$ matrix of indeterminants. For any ring $R$, let

\end{convention}

\begin{definition}Let $V$ be a module over an arbitrary ring $R$. Suppose $\sigma:R\rightarrow R$ is a ring map satisfying $\sigma\circ\sigma=\mathrm{id}$, which we call an \textit{involution} map. A function
    \[b:V\times V\rightarrow R\]
    is called a \textit{form} on $V$, and is said to be
    \begin{enumerate}
        \item \textit{left linear} if $b(rv+u,w)=rb(v,w)+b(u,w)$ for all $r\in R$ and $u,v,w\in V$,
        \item \textit{left $\sigma$-linear} if $b(rv+u,w)=\sigma(r)b(v,w)+b(u,w)$ for all $r\in R$ and $u,v,w\in V$,
        \item \textit{right linear} if $b(v,rw+u)=rb(v,w)+b(v,u)$ for all $r\in R$ and $u,v,w\in V$,
        \item \textit{bilinear} if it is left linear and right linear,
        \item \textit{$\sigma$-sesquilinear} if it is left $\sigma$-linear and right linear,
        \item \textit{symmetric} if $b(v, w)=b(w, v)$ for all $v,w\in V$,
        \item \textit{$\sigma$-symmetric} if $b(v, w)=\sigma(b(w, v))$ for all $v,w\in V$,
        \item \textit{alternating} if $b(v, v)=0$ for all $v\in V$,
        \item \textit{nondegenerate} if $b(v, w)=0$ for all $v\in V$ implies $w=0$,
        \item \textit{orthogonal} if it is a bilinear, symmetric, and nondegenerate,
        \item \textit{symplectic} if it is a bilinear, alternating, and nondegenerate,
        \item \textit{$\sigma$-Hermitian} if it is $\sigma$-sesquilinear, $\sigma$-symmetric, and nondegenerate.
    \end{enumerate}
\end{definition}
\begin{definition}Let $b_1:V\times V\rightarrow R$ and $b_2:V\times V\rightarrow R$ be forms on a module $V$ over an arbitrary ring $R$. We say $b_1$ is \textit{equivalent} (or \textit{isometric}) to $b_2$ if there exists an invertible linear map $P:V\rightarrow V$, which we call an \textit{isometry}, satisfying $b_1(v,w)=b_2(P(v),P(w))$ for all $v,w\in V$. 
\end{definition}

\begin{example}Let $V=R^n$ be the free module of rank $n$ over a ring $R$. The \textit{standard orthogonal form} of signature $(p,q)$ on $V$, where $p,q\in\mathbb N$ such that $p+q=n$, is 
    \[b_{p,q}:V\times V\rightarrow E\qquad (x,y)\mapsto \underbrace{x_1y_1+\cdots+x_py_p}_p-\overbrace{(x_{p+1}y_{p+1}+\cdots+x_ny_n)}^q\]
    where $b_{n,0}=\langle\cdot,\cdot\rangle$ is the \textit{standard inner product}. 
\end{example}

\begin{example}Recall that a quadratic form on a free module $V=R^n$ of rank $n$ over $R$ is a quadratic homogeneous polynomial map $q:V\rightarrow R$ 
    \[q(x_1,\dots,x_n)=\sum_{1\le i,j\le n}a_{i,j}x_ix_j\]
    Recall the polar form $b_q:V\times V\rightarrow R$ of a quadratic form $q:V\rightarrow R$ is the symmetric bilinear form 
    \[b_q(x,y)=q(x+y)-q(x)-q(y)\]
    and we say $q$ is nonsingular if $b_q$ is nondegenerate. Suppose $2$ is invertible in $R$, then we can define the polarization $b=\frac{1}{2}b_q$ of $q$, which satisfies $q(x)=b(x,x)$. 
    Therefore, when $2$ is invertible in $R$, a quadratic form corresponds uniquely to a symmetric bilinear form. 
\end{example}

\begin{definition}
    \label{def:quadetale}
    A \textit{quadratic étale algebra} $E$ over a ring $R$ is a $R$-algebra such that \[E\cong R[\omega]/(\omega^2+b\omega+c)\]
    for some $b,c\in R$ where $b^2-4c\in R^\times$. Define its \textit{involution} as the unique nontrivial automorphism 
    \[\sigma:E\rightarrow E\qquad \omega\mapsto -b-\omega\]
    which satisfies $\sigma\circ \sigma=\mathrm{id}_E$ and fixes $R$. 
\end{definition}
\begin{example}Here are some examples of quadratic étale algebras
    \begin{enumerate}
        \item $\mathbb C$ as a $\mathbb R$-algebra, with $\sigma:\mathbb C\rightarrow\mathbb C$ given by $\sigma(z)=\overline{z}$
        \item any quadratic extension $K(\sqrt{d})$ of a field $K$, with $\sigma:K(\sqrt{d})\rightarrow K(\sqrt{d})$ by $\sqrt{d}\mapsto -\sqrt{d}$
        \item $\mathbb F_{q^2}$ as a $\mathbb F_{q}$-algebra, with $\sigma:\mathbb F_{q^2}\rightarrow\mathbb F_{q^2}$ given by $\sigma(x)=x^q$
    \end{enumerate}
\end{example}

    where $h_{n,0}$ is just known as the \textit{standard $\sigma$-Hermitian form}. 
\end{example}

        \item the \textit{unitary group} of the pair $(V,h)$
        \[\mathrm{U}_{V,h}:\mathsf{Alg}_R\rightarrow \mathsf{Grp}\]
        where for each $R$-algebra $S$
        \[\mathrm{U}_{V,h}(S)=\{g\in\mathrm{Aut}_{\mathsf{Mod}_S}(V_{E_S}):\forall v,w\in V_{E_S}, h_S(gv,gw)=h_S(v,w)\}\]
        where for indeterminant matrices $X=(x_{i,j})_{1\le i,j\le n}$ and $Y=(y_{i,j})_{1\le i,j\le n}$ 
        \[\widetilde{\mathrm{U}_{V,h}}=\mathrm{Spec}\left(\frac{R[X,Y][y]}{((X+Y\omega)^*[h]_e(X+Y\omega)-[h]_e,y\det(X+Y\omega)-1)_{1,\omega}}\right)\]
        where $e$ is an ordered basis of $V$, on which $\mathrm{U}_{V,h}$ does not depend, 
        \item the \textit{special unitary group} of the pair $(V,h)$
        \[\mathrm{SU}_{V,h}:\mathsf{Alg}_R\rightarrow \mathsf{Grp}\]
        where for each $R$-algebra $S$
        \[\mathrm{SU}_{V,h}(S)=\{g\in\mathrm{U}_{V,h}(S):\det(g)=1\}\]
        where for indeterminant matrices $X=(x_{i,j})_{1\le i,j\le n}$ and $Y=(y_{i,j})_{1\le i,j\le n}$ 
        \[\widetilde{\mathrm{SU}_{V,h}}=\mathrm{Spec}\left(\frac{R[X,Y]}{((X+Y\omega)^*[h]_e(X+Y\omega)-[h]_e,\det(X+Y\omega)-1)_{1,\omega}}\right)\]
        where $e$ is an ordered basis of $V$, on which $\mathrm{SU}_{V,h}$ does not depend, 
        \item the \textit{unitary semilitude group} of the pair $(V,h)$
        \[\mathrm{GU}_{V,h}:\mathsf{Alg}_R\rightarrow \mathsf{Grp}\]
        where for each $R$-algebra $S$
        \[\mathrm{GU}_{V,h}(S)=\{g\in\mathrm{Aut}_{\mathsf{Mod}_S}(V_{E_S}):\exists\lambda\in S^\times,\forall v,w\in V_{E_S}, h_S(gv,gw)=(1\otimes\lambda) h_S(v,w)\}\]
        where for indeterminant matrices $X=(x_{i,j})_{1\le i,j\le n}$ and $Y=(y_{i,j})_{1\le i,j\le n}$ 
        \[\widetilde{\mathrm{GU}_{V,h}}=\mathrm{Spec}\left(\frac{R[X,Y][y,\lambda,\nu]}{((X+Y\omega)^*[h]_e(X+Y\omega)-\lambda[h]_e,y\det(X+Y\omega)-1, \lambda\nu-1)_{1,\omega}}\right)\]
        where $e$ is an ordered basis of $V$, on which $\mathrm{GU}_{V,h}$ does not depend, 
        \item the \textit{special unitary semilitude group} of the pair $(V,h)$
        \[\mathrm{GSU}_{V,h}:\mathsf{Alg}_R\rightarrow \mathsf{Grp}\]
        where for each $R$-algebra $S$
        \[\mathrm{GSU}_{V,h}(S)=\{g\in\mathrm{GU}_{V,h}(S):\det(g)=1\}\]
        where for indeterminant matrices $X=(x_{i,j})_{1\le i,j\le n}$ and $Y=(y_{i,j})_{1\le i,j\le n}$ 
        \[\widetilde{\mathrm{GSU}_{V,h}}=\mathrm{Spec}\left(\frac{R[X,Y][\lambda,\nu]}{((X+Y\omega)^*[h]_e(X+Y\omega)-\lambda[h]_e,\det(X+Y\omega)-1, \lambda\nu-1)_{1,\omega}}\right)\]
        where $e$ is an ordered basis of $V$, on which $\mathrm{GSU}_{V,h}$ does not depend, 
        
    \end{enumerate}
\end{example}
\begin{remark}The choice of $\omega$ for a quadratic étale $R$-algebra $E$ is not necessarily unique, but one can show that none of the groups in \Cref{ex:unitary} depend on the choice of $\omega$. 
\end{remark}

\begin{definition}
Let $V$ be a free module over $R$ and $q:V\rightarrow R$ a quadratic form. The \textit{Clifford algebra} $\mathrm{Cl}(V,q)$ is the universal associative unital $R$-algebra generated by $V$ and satisfying $v^2=q(v)1$, in the sense that for any unital associative $R$-algebra $C$ and $R$-linear map $f:V\rightarrow C$ such that $f(v)^2=q(v)1$, there exists unique $R$-algebra morphism $\widetilde{f}:\mathrm{Cl}(V,q)\rightarrow C$ such that $f=\widetilde{f}\circ i$ where $i:V\rightarrow\mathrm{Cl}(V,q)$ is the canonical inclusion. 
By the universal property, the map $V\rightarrow\mathrm{Cl}(V,q)$ by $v\mapsto -v$ extends to a map $\alpha:\mathrm{Cl}(V,q)\rightarrow\mathrm{Cl}(V,q)$ called the \textit{involution}, and therefore we have a canonical $\mathbb Z/2$-grading on the Clifford algebra $\mathrm{Cl}(V,q)=\mathrm{Cl}^0(V,q)\otimes \mathrm{Cl}^1(V,q)$ where 
\[\begin{aligned}
    \mathrm{Cl}^0(V,q)&=\{x\in \mathrm{Cl}(V,q):\alpha(x)=x\} \\
    \mathrm{Cl}^1(V,q)&=\{x\in \mathrm{Cl}(V,q):\alpha(x)=-x\}\\
\end{aligned}\]
making the Clifford algebra a $\mathbb Z/2$-graded algebra. 
\end{definition}

\begin{remark}
    Let $V$ be a free module over $R$ and $q:V\rightarrow R$ a quadratic form. Constructively, one can define the Clifford algebra to be
    \[\mathrm{Cl}(V,q)=\bigoplus_{i\in\mathbb N}R^{\otimes i}\big/(v\otimes v-q(v)1)\]
    from which the $\mathbb Z/2$-grading is inherited. 
\end{remark}
\begin{definition}Let $V$ be a free module over $R$ and $q:V\rightarrow R$ a quadratic form. 
    Let $\mathrm{Cl}(V,q)^{\mathrm{op}}$ be the opposite algebra of the Clifford alegbra, i.e. it has the same underlying set and additive structure but with $xy$ defined to be $yx$ in $\mathrm{Cl}(V,q)$. By the universal property, the map $V\rightarrow\mathrm{Cl}(V,q)^{\mathrm{op}}$ given by $v\mapsto -v$ extends to a map $c:\mathrm{Cl}(V,q)\rightarrow \mathrm{Cl}(V,q)^{\mathrm{op}}$, which after composing with the canonical identification $\mathrm{Cl}(V,q)^{\mathrm{op}}\cong \mathrm{Cl}(V,q)$ gives a function 
    \[\overline{(\cdot)}:\mathrm{Cl}(V,q)\rightarrow \mathrm{Cl}(V,q)\qquad x\mapsto \overline{x}\]
    which we call the \textit{Clifford conjugation}. 
\end{definition}
\begin{remark}Let $V$ be a free module over $R$ and $q:V\rightarrow R$ a quadratic form. Let $S$ be an $R$-algebra, we write the base change $\mathrm{Cl}_S(V,q)=\mathrm{Cl}(V,q)\otimes_R S\cong \mathrm{Cl}(V_S,q_S)$, where $q_S:V_S\rightarrow S$ is the base change of the quadratic form defined by $(v\otimes s)\mapsto q(v)s^2$. 
\end{remark}
\begin{example}Let $V$ be a free module over $R$ and $q:V\rightarrow R$ a quadratic form. Here are some more examples of algebraic groups over $R$. 
    \begin{enumerate}
        \item the \textit{Clifford group} or \textit{similitude pin group} of the pair $(V,q)$
        \[\mathrm{GPin}_{V,q}:\mathsf{Alg}_R\rightarrow\mathsf{Grp}\]
        where 
        \[\mathrm{GPin}_{V,q}(S)=\{x\in \mathrm{Cl}_R(V,q)^\times:xV_Sx^{-1}\subseteq V_S\}\]
        for each $R$-algebra $S$,
        \item the \textit{pin group} of the pair $(V,q)$
        \[\mathrm{Pin}_{V,q}:\mathsf{Alg}_R\rightarrow\mathsf{Grp}\]
        where 
        \[\mathrm{Pin}_{V,q}(S)=\{x\in\mathrm{GPin}_{V,q}(S):x\overline{x}=1\}\]
        for each $R$-algebra $S$,

        \item the \textit{similitude spin group} of the pair $(V,q)$
        \[\mathrm{GSpin}_{V,q}:\mathsf{Alg}_R\rightarrow\mathsf{Grp}\]
        where 
        \[\mathrm{GSpin}_{V,q}(S)=\mathrm{GPin}_{V,q}(S)\cap \mathrm{Cl}^0(V,q)\]
        for each $R$-algebra $S$,

        \item the \textit{spin group} of the pair $(V,q)$
        \[\mathrm{Spin}_{V,q}:\mathsf{Alg}_R\rightarrow\mathsf{Grp}\]
        where 
        \[\mathrm{Spin}_{V,q}(S)=\{x\in \mathrm{GSpin}_{V,q}(S):x\overline{x}=1\}\]
        for each $R$-algebra $S$.
        
    \end{enumerate}
\end{example}

\begin{remark}Another important family of algebraic groups is the \textit{exceptional groups}: $\mathrm G_2, \mathrm F_4, \mathrm E_6, \mathrm E_7$, and $\mathrm E_8$, which we will not define until later. 
\end{remark}
\begin{definition}
    \label{def:base}
    Let $G$ be an algebraic group over $R$ and $S$ a finite projective $R$-algebra. Define
    \begin{enumerate}
        \item \textit{base change functor} $(-)_S:\mathsf{GAlg}_R\rightarrow \mathsf{GAlg}_S$ given by $G_S(T)=G(T)$ for any $S$-algebra $T$, and maps morphisms to themselves forgetfully;
        \item \textit{Weil restriction functor} $(-)_R:\mathsf{GAlg}_S\rightarrow \mathsf{GAlg}_R$ given by $G_R(T)=G(T\otimes_R S)$ for any $R$-algebra $T$, in other words, $G_R=G\circ (-\otimes_R S)$, and for a morphism $\varphi:G\rightarrow H$ of algebraic groups over $S$, the induced map $\varphi_R:G_R\rightarrow H_R$ is $(\varphi_R)_T=\varphi_{T\otimes_R S}$ for each $R$-algebra $T$. 
    \end{enumerate}
\end{definition}
\begin{remark}Recall that we say a $R$-algebra $S$ is finite projective if there exists a $R$-module $T$ such that $S\oplus T$ is a free module of finite rank. The assumption of finite projectiveness in \Cref{def:base} is to ensure the base changed or restricted group functors are indeed algebraic groups. 
\end{remark}
